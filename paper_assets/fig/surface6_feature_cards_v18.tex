% --- Inline lexicon highlight (variant B: soft highlighter mark, no border) ---
% Borderless, flat tint deepened ~12% toward the accent; reads as a subtle
% highlighter swipe behind the word, clearly softer than the stroked varname pills.
\newcommand{\miniptok}[3]{%
  \tikz[baseline=(t.base)]{%
    \node[rounded corners=1.6pt, draw=black!17, line width=0.4pt, fill=black!3,
          inner xsep=2.6pt, inner ysep=0.9pt, text height=5.6pt, text depth=1.9pt,
          anchor=base, outer sep=0pt] (t)
          {\textcolor{Body}{\ttfamily\fontsize{8.2pt}{9.2pt}\selectfont #3}};
  }%
}
\newcommand{\featchip}[3]{% #1 accent  #2 num  #3 varname
  \tikz[baseline=(f.base)]{%
    \node[rounded corners=2.6pt, draw=none, fill=#1!88!white,
          inner xsep=4.6pt, inner ysep=2.1pt, text height=5.8pt, text depth=2.2pt,
          anchor=base, outer sep=0pt] (f)
          {\textcolor{white}{\ttfamily\bfseries\fontsize{7.2pt}{8.4pt}\selectfont #2}%
           \hspace{3.2pt}{\color{white!40!#1}\vrule width 0.5pt height 5.2pt depth 1.6pt}\hspace{3.2pt}%
           \textcolor{white}{\ttfamily\bfseries\fontsize{8.4pt}{10pt}\selectfont #3}};
  }%
}

\newcommand{\toksep}{\hspace{2.4pt plus 1pt minus 0.6pt}}

% --- Definitions ---
\newcommand{\defNegLead}{Indicator equal to $1$ if the answer begins with any of $\{$\kern-0.6pt\negtok{no}\toksep\negtok{not}\toksep\negtok{never}\toksep\negtok{none}\toksep
  \negtok{nothing}\toksep\negtok{nobody}\toksep\negtok{there is no}\toksep
  \negtok{there are no}\toksep\negtok{there's no}\kern-0.6pt$\}$; otherwise $0$.}
\newcommand{\defNegCount}{Total occurrences of $\{$\kern-0.6pt\negtok{no}\toksep\negtok{not}\toksep\negtok{never}\toksep\negtok{none}\toksep
  \negtok{nothing}\toksep\negtok{nobody}\toksep\negtok{cannot}\toksep
  \negtok{can't}\toksep\negtok{don't}\toksep\negtok{doesn't}\toksep
  \negtok{didn't}\toksep\negtok{won't}\toksep\negtok{wasn't}\toksep
  \negtok{weren't}\kern-0.6pt$\}$.}
\newcommand{\defHedgeRate}{Occurrences of $\{$\kern-0.6pt\hedgetok{might}\toksep\hedgetok{could}\toksep\hedgetok{may}\toksep
  \hedgetok{generally}\toksep\hedgetok{typically}\toksep\hedgetok{often}\toksep
  \hedgetok{usually}\toksep\hedgetok{sometimes}\toksep\hedgetok{likely}\toksep
  \hedgetok{probably}\toksep\hedgetok{possibly}\toksep\hedgetok{appears to}\toksep
  \hedgetok{seems}\toksep\hedgetok{suggests}\toksep\hedgetok{roughly}\toksep
  \hedgetok{approximately}\toksep\hedgetok{are expected to}\kern-0.6pt$\}$, divided by word count.}
\newcommand{\defWordCount}{Number of whitespace-separated tokens: \texttt{len(text.split())}.}
\newcommand{\defMeanToken}{Mean character length over alphanumeric tokens.}
\newcommand{\defTTR}{$\lvert\text{unique tokens}\rvert/\lvert\text{total tokens}\rvert$; closer to $1$ means higher lexical diversity.}

  {\hyphenpenalty=10000\exhyphenpenalty=10000\spaceskip=0.30em\relax\fontsize{7.8pt}{8.4pt}\selectfont\bfseries\color{#2!85!black}%

  \hspace{0.6em}%
  {\color{#2!45}\leaders\hrule height 0.6pt\hfill\kern0pt}%
  \par\vspace{0.30em}%
}
